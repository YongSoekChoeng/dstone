```sql
-- [TS] 이미지게시판 - 이미지 첨부 게시물(삭제는 use_yn='N')
CREATE TABLE tbts_image_board (
    seq        BIGINT        NOT NULL AUTO_INCREMENT,
    mb_id      VARCHAR(20)   NOT NULL,
    title      VARCHAR(200)  NOT NULL,
    writer_nm  VARCHAR(50)   NULL,
    file_no    VARCHAR(40)   NULL COMMENT '첨부파일번호(tbsy_file_info.file_no)',
    use_yn     VARCHAR(1)    NOT NULL DEFAULT 'Y',
    in_emp_cd  VARCHAR(20)   NULL,
    in_dtm     DATETIME      NULL,
    up_emp_cd  VARCHAR(20)   NULL,
    up_dtm     DATETIME      NULL,
    PRIMARY KEY (seq)
);
```
