# DDL - 이미지 게시판

## 1. 테이블 신설

```sql
CREATE TABLE tbts_image_board (
  seq        BIGINT(19)   NOT NULL AUTO_INCREMENT COMMENT '시퀀스',
  mb_id      VARCHAR(20)  NOT NULL COMMENT '회사코드',
  title      VARCHAR(200) NOT NULL COMMENT '제목',
  writer_nm  VARCHAR(50)  NULL COMMENT '등록자명',
  image_url  VARCHAR(500) NULL COMMENT '이미지 URL',
  thumb_url  VARCHAR(500) NULL COMMENT '썸네일 URL',
  use_yn     VARCHAR(1)   NOT NULL DEFAULT 'Y' COMMENT '사용여부',
  in_emp_cd  VARCHAR(20)  NULL COMMENT '등록사번',
  in_dtm     DATETIME     NULL COMMENT '등록일시',
  up_emp_cd  VARCHAR(20)  NULL COMMENT '수정사번',
  up_dtm     DATETIME     NULL COMMENT '수정일시',
  PRIMARY KEY (seq)
) COMMENT='이미지 게시판';
```
