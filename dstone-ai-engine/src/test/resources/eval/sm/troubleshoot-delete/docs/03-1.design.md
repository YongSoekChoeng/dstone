# 설계서 - 이미지 게시판

## 설계 요약
- 샘플게시판(kr.co.gnx.sample.board)의 층별 파일(화면/Controller/Service/DAO/VO/매퍼)을 참고로, 새 패키지 kr.co.gnx.test.imageboard에 이미지 게시판 목록화면과 상세화면(팝업)을 신규 구현한다.
- 새 테이블 tbts_image_board(제목/등록자/이미지URL)를 신설하고, tbsy_resource에 '테스트'(TS_000) 하위 메뉴 '이미지게시판'을 등록한다.
- 목록화면은 Kendo 그리드(체크박스/SEQ/썸네일/제목/등록자, 서버 페이징, 제목 조회)이며, 제목 클릭→상세 팝업(조회모드), 썸네일 클릭→이미지 팝업, 전체선택/등록/삭제 버튼을 둔다.
- 상세화면은 genexon.PopWindowOpen으로 여는 팝업(편집/조회 모드, 전체화면의 70% 크기)이며, 파일 업로드는 기존 FileService.imageSave(MultipartFile, UPLOADS.UPIMAGE)를 그대로 쓰고, 원본에서 60x60 썸네일을 자동 생성한다.
- 기존 파일은 하나도 수정하지 않는다(신규 파일 8개 + DDL 1 + 메뉴 INSERT 1).

### DDL (사람이 DB에 반영)

```sql
-- 이미지 게시판 테이블
CREATE TABLE tbts_image_board (
  seq        BIGINT(19)   NOT NULL AUTO_INCREMENT COMMENT '시퀀스',
  mb_id      VARCHAR(20)  NOT NULL COMMENT '회사코드',
  title      VARCHAR(200) NOT NULL COMMENT '제목',
  writer_nm  VARCHAR(50)  NULL COMMENT '등록자명',
  image_url  VARCHAR(500) NULL COMMENT '이미지 URL',
  thumb_url  VARCHAR(500) NULL COMMENT '썸네일 URL',
  use_yn     VARCHAR(1)   NOT NULL DEFAULT 'Y' COMMENT '사용여부',
  in_emp_cd  VARCHAR(20)  NULL COMMENT '등록사번',
  in_dtm     DATETIME     NULL COMMENT '등록일시',
  up_emp_cd  VARCHAR(20)  NULL COMMENT '수정사번',
  up_dtm     DATETIME     NULL COMMENT '수정일시',
  PRIMARY KEY (seq)
) COMMENT='이미지 게시판';
```

#### 3. ImageBoardService.java (전체)
```java
package kr.co.gnx.test.imageboard;

import java.awt.Graphics2D;
import java.awt.RenderingHints;
import java.awt.image.BufferedImage;
import java.io.File;
import java.util.List;
import java.util.Map;

import javax.imageio.ImageIO;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import kr.co.gnx.base.BaseService;
import kr.co.gnx.comm.util.CommUtil;
import kr.co.gnx.config.Constants.UPLOADS;
import kr.co.gnx.system.file.FileService;
import kr.co.gnx.system.file.FileVO;

/**
 * 이미지 게시판 서비스
 * 이름이 *Service 라서 context-transaction.xml 포인트컷에 걸린다 - 메서드 하나가 트랜잭션 하나.
 */
@Service
public class ImageBoardService extends BaseService {
	private static final Logger logger = LoggerFactory.getLogger(ImageBoardService.class);

	@Autowired
	private ImageBoardDAO imageBoardDAO;

	@Autowired
	private FileService fileService;

	public List<Map<String, Object>> getImageBoardList(ImageBoardVO vo) {
		return imageBoardDAO.selectImageBoardList(vo);
	}

	public Map<String, Object> getImageBoardView(ImageBoardVO vo) {
		return imageBoardDAO.selectImageBoardView(vo);
	}

	/**
	 * 등록/수정. 파일이 있으면 이미지 저장 후 URL을 VO에 채운다.
	 */
	public int saveImageBoard(MultipartFile file, ImageBoardVO vo) throws Exception {
		if (file != null && !file.isEmpty()) {
			FileVO fileVO = fileService.imageSave(file, UPLOADS.UPIMAGE);
			vo.setImage_url(fileVO.getFile_url());
			vo.setThumb_url(makeThumbnail(fileVO));
		}
		if (CommUtil.isEmpty(vo.getSeq())) {
			return imageBoardDAO.insertImageBoard(vo);
		} else {
			return imageBoardDAO.updateImageBoard(vo);
		}
	}

	public int deleteImageBoard(List<Long> seqList) {
		int cnt = 0;
		for (Long seq : seqList) {
			ImageBoardVO vo = new ImageBoardVO();
			vo.setSeq(seq.toString());
			cnt += imageBoardDAO.deleteImageBoard(vo);
		}
		return cnt;
	}

	/**
	 * 원본 이미지를 그리드 썸네일 컬럼(폭 100)에 맞춰 60x60으로 리사이즈한 썸네일을 생성.
	 * 생성 실패 시 원본 URL을 돌려준다.
	 */
	private String makeThumbnail(FileVO fileVO) {
		int size = 60; // 그리드 썸네일 컬럼(폭 100)을 감안해 자체계산
		try {
			BufferedImage img = ImageIO.read(new File(fileVO.getFile_path()));
			BufferedImage thumb = new BufferedImage(size, size, BufferedImage.TYPE_INT_RGB);
			Graphics2D g = thumb.createGraphics();
			g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
			g.drawImage(img, 0, 0, size, size, null);
			g.dispose();
			String dir = fileVO.getFile_path().substring(0, fileVO.getFile_path().lastIndexOf(File.separatorChar));
			String name = "thumb_" + fileVO.getFile_no() + "." + fileVO.getFile_format();
			ImageIO.write(thumb, fileVO.getFile_format(), new File(dir + File.separator + name));
			return fileVO.getFile_url().substring(0, fileVO.getFile_url().lastIndexOf("/")) + "/" + name;
		} catch (Exception e) {
			logger.warn("썸네일 생성 실패, 원본 표시: " + e.getMessage());
			return fileVO.getFile_url();
		}
	}
}
```

#### 4. ImageBoardController.java (전체)
```java
package kr.co.gnx.test.imageboard;

import java.util.ArrayList;
import java.util.List;

import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.multipart.MultipartFile;
import org.springframework.web.multipart.MultipartHttpServletRequest;
import org.springframework.web.servlet.ModelAndView;

import kr.co.gnx.base.BaseController;
import kr.co.gnx.comm.util.CommUtil;
import kr.co.gnx.comm.util.session.SessionUtil;
import kr.co.gnx.security.model.User;
import net.sf.json.JSONArray;
import net.sf.json.JSONObject;

/**
 * 이미지 게시판 (테스트 메뉴)
 * 목록화면 + 상세화면(팝업, 편집/조회 모드).
 * 등록/수정은 multipart 폼(제목 + 이미지 파일), 삭제는 seq 배열 JSON.
 */
@Controller(value="ImageBoardController")
public class ImageBoardController extends BaseController {
	private static final Logger logger = LoggerFactory.getLogger(ImageBoardController.class);

	@Autowired
	private ImageBoardService imageBoardService;

	/**
	 * @brief 이미지 게시판 목록화면 이동
	 */
	@RequestMapping(value = "/test/imageboard/imageBoard.go")
	public ModelAndView imageBoard(HttpServletRequest request, HttpServletResponse response) throws Exception {
		return new ModelAndView("/test/imageboard/imageBoard");
	}

	/**
	 * @brief 상세화면 팝업 이동 (seq 있으면 조회/수정 모드, 없으면 편집 모드)
	 */
	@RequestMapping(value = "/test/imageboard/imageBoardDetail.pop")
	public ModelAndView imageBoardDetail(HttpServletRequest request, HttpServletResponse response, ImageBoardVO vo) throws Exception {
		ModelAndView mv = new ModelAndView("/test/imageboard/imageBoardDetail");
		if (CommUtil.isNotEmpty(vo.getJson_string())) {
			JSONObject paramJObj = JSONObject.fromObject(vo.getJson_string());
			vo = (ImageBoardVO) JSONObject.toBean(paramJObj, ImageBoardVO.class);
		}
		if (vo.getSeq() != null) {
			vo.setMb_id(getLoginUser(request).getMb_id());
			mv.addObject("ImageBoardVO", imageBoardService.getImageBoardView(vo));
		} else {
			mv.addObject("ImageBoardVO", vo);
		}
		return mv;
	}

	/**
	 * @brief 목록 조회 (search_word: 제목 검색어)
	 */
	@RequestMapping(value = "/test/imageboard/getImageBoardList.ajax")
	public ModelAndView getImageBoardList(HttpServletRequest request, HttpServletResponse response, ImageBoardVO vo) throws Exception {
		ModelAndView mv = new ModelAndView("jsonView");

		// 그리드는 검색조건을 json_string 하나로 묶어 보낸다
		if(CommUtil.isNotEmpty(vo.getJson_string())) {
			JSONObject paramJObj = JSONObject.fromObject(vo.getJson_string());
			vo = (ImageBoardVO) JSONObject.toBean(paramJObj, ImageBoardVO.class);
		}
		vo.setMb_id(getLoginUser(request).getMb_id());

		mv.addObject("results", imageBoardService.getImageBoardList(vo));
		return mv;
	}

	/**
	 * @brief 단건 조회 (seq 필수)
	 */
	@RequestMapping(value = "/test/imageboard/getImageBoardView.ajax")
	public ModelAndView getImageBoardView(HttpServletRequest request, HttpServletResponse response, ImageBoardVO vo) throws Exception {
		ModelAndView mv = new ModelAndView("jsonView");
		vo.setMb_id(getLoginUser(request).getMb_id());

		mv.addObject("result", imageBoardService.getImageBoardView(vo));
		return mv;
	}

	/**
	 * @brief 등록/수정 (multipart 폼: title, image_file. seq 있으면 수정)
	 */
	@RequestMapping(value = "/test/imageboard/saveImageBoard.ajax")
	public ModelAndView saveImageBoard(MultipartHttpServletRequest request, ImageBoardVO vo) throws Exception {
		ModelAndView mv = new ModelAndView("jsonView");
		User user = getLoginUser(request);

		MultipartFile file = request.getFile("image_file");
		vo.setMb_id(user.getMb_id());
		vo.setIn_emp_cd(user.getEmp_cd());
		vo.setUp_emp_cd(user.getEmp_cd());
		if(CommUtil.isEmpty(vo.getWriter_nm())) {
			vo.setWriter_nm(user.getEmp_nm());
		}

		mv.addObject("count", imageBoardService.saveImageBoard(file, vo));
		return mv;
	}

	/**
	 * @brief 삭제 (JSON 본문: {"seqList":[seq,...]}, 물리 삭제)
	 */
	@RequestMapping(value = "/test/imageboard/deleteImageBoard.ajax")
	public ModelAndView deleteImageBoard(HttpServletRequest request, HttpServletResponse response, @RequestBody JSONObject paramJObj) throws Exception {
		ModelAndView mv = new ModelAndView("jsonView");
		User user = getLoginUser(request);

		JSONArray seqArr = paramJObj.getJSONArray("seqList");
		List<Long> seqList = new ArrayList<Long>();
		for (int i = 0; i < seqArr.size(); i++) {
			seqList.add(seqArr.getLong(i));
		}

		mv.addObject("count", imageBoardService.deleteImageBoard(seqList));
		return mv;
	}

	private User getLoginUser(HttpServletRequest request) {
		return SessionUtil.getSessionVO(request.getSession()).getUser();
	}
}
```

#### 5. imageboard-mapper.xml (전체)
```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<!--
	이미지 게시판(tbts_image_board) - 테스트 메뉴.
	삭제는 물리 DELETE로 행을 지운다.
-->
<mapper namespace="ImageBoard">
	<select id="selectImageBoardList" parameterType="kr.co.gnx.test.imageboard.ImageBoardVO" resultType="bmap">
		/* imageboard-mapper.xml id="selectImageBoardList" 이미지 게시판 목록 조회 */
		SELECT seq
		     , mb_id
		     , title
		     , writer_nm
		     , image_url
		     , thumb_url
		     , DATE_FORMAT(in_dtm, '%Y-%m-%d %H:%i:%s') AS in_dtm
		     , DATE_FORMAT(up_dtm, '%Y-%m-%d %H:%i:%s') AS up_dtm
		  FROM tbts_image_board
		 WHERE mb_id = #{mb_id}
		   AND use_yn = 'Y'
		<if test="@kr.co.gnx.comm.util.CommUtil@isNotEmpty(search_word)">
		   AND title LIKE CONCAT('%', #{search_word}, '%')
		</if>
		 ORDER BY seq DESC
	</select>

	<select id="selectImageBoardView" parameterType="kr.co.gnx.test.imageboard.ImageBoardVO" resultType="bmap">
		/* imageboard-mapper.xml id="selectImageBoardView" 이미지 게시판 단건 조회 */
		SELECT seq
		     , mb_id
		     , title
		     , writer_nm
		     , image_url
		     , thumb_url
		     , DATE_FORMAT(in_dtm, '%Y-%m-%d %H:%i:%s') AS in_dtm
		     , DATE_FORMAT(up_dtm, '%Y-%m-%d %H:%i:%s') AS up_dtm
		  FROM tbts_image_board
		 WHERE mb_id = #{mb_id}
		   AND seq   = #{seq}
		   AND use_yn = 'Y'
	</select>

	<insert id="insertImageBoard" parameterType="kr.co.gnx.test.imageboard.ImageBoardVO">
		/* imageboard-mapper.xml id="insertImageBoard" 이미지 게시판 등록 */
		INSERT
		  INTO tbts_image_board
		     ( mb_id
		     , title
		     , writer_nm
		     , image_url
		     , thumb_url
		     , use_yn
		     , in_emp_cd
		     , in_dtm
		     )
		VALUES
		     ( #{mb_id}
		     , #{title}
		     , #{writer_nm}
		     , #{image_url}
		     , #{thumb_url}
		     , 'Y'
		     , #{in_emp_cd}
		     , NOW()
		     )
	</insert>

	<update id="updateImageBoard" parameterType="kr.co.gnx.test.imageboard.ImageBoardVO">
		/* imageboard-mapper.xml id="updateImageBoard" 이미지 게시판 수정 */
		UPDATE tbts_image_board
		   SET title     = #{title}
		     , writer_nm = #{writer_nm}
		     , image_url = #{image_url}
		     , thumb_url = #{thumb_url}
		     , up_emp_cd = #{up_emp_cd}
		     , up_dtm    = NOW()
		 WHERE mb_id = #{mb_id}
		   AND seq   = #{seq}
	</update>

	<delete id="deleteImageBoard" parameterType="kr.co.gnx.test.imageboard.ImageBoardVO">
		/* imageboard-mapper.xml id="deleteImageBoard" 이미지 게시판 삭제(물리 삭제) */
		DELETE FROM tbts_image_board
		 WHERE mb_id = #{mb_id}
		   AND seq   = #{seq}
	</delete>
</mapper>
```
